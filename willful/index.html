/**
 * Willful month-end autopilot (Google Apps Script)
 * Sheets needed: "Willful orders" [orderId, partner, completedAt]
 *                "Partner reports" [orderId, partner]
 *                "Partners" [partner, contactName, email, feePerPlan]
 * Writes a "Reconciliation" sheet, creates invoice rows, and
 * drafts (not sends) a Gmail follow-up for any partner over 2%.
 * Trigger: time-driven, 1st of the month, 6am.
 */
function runMonthEnd() {
  const ss = SpreadsheetApp.getActive();
  const rows = name => ss.getSheetByName(name).getDataRange().getValues().slice(1);
  const partners = Object.fromEntries(rows('Partners').map(r => [r[0], {name: r[1], email: r[2], fee: r[3]}]));
  const ours = {}, theirs = {};
  rows('Willful orders').forEach(([id, p]) => (ours[p] = ours[p] || new Set()).add(String(id)));
  rows('Partner reports').forEach(([id, p]) => (theirs[p] = theirs[p] || new Set()).add(String(id)));

  const out = [['Partner', 'Partner says', 'Willful recorded', 'Matched', 'Gap %', 'Status', 'Invoice']];
  Object.keys(partners).forEach(p => {
    const a = ours[p] || new Set(), b = theirs[p] || new Set();
    const matched = [...a].filter(id => b.has(id));
    const onlyTheirs = [...b].filter(id => !a.has(id));
    const onlyOurs = [...a].filter(id => !b.has(id));
    const gap = b.size ? Math.abs(b.size - a.size) / b.size * 100 : 0;
    const flagged = gap > 2 || onlyTheirs.length + onlyOurs.length > 5;
    out.push([p, b.size, a.size, matched.length, gap.toFixed(1), flagged ? 'CHECK' : 'OK', matched.length * partners[p].fee]);
    if (flagged) GmailApp.createDraft(partners[p].email,
      `November numbers: ${onlyTheirs.length + onlyOurs.length} orders to check together`,
      `Hi ${partners[p].name},\n\nThanks for sending November. We match on ${matched.length} plans.\n` +
      `On your side only (${onlyTheirs.length}): ${onlyTheirs.join(', ')}\n` +
      `On our side only (${onlyOurs.length}): ${onlyOurs.join(', ')}\n\n` +
      `Could you check these on your end? Once we agree I'll send the invoice the same day.\n\nThanks,\nDurva`);
  });
  const sh = ss.getSheetByName('Reconciliation') || ss.insertSheet('Reconciliation');
  sh.clear().getRange(1, 1, out.length, out[0].length).setValues(out);
}

"Are online wills actually legal?"Yes. A Willful will is legal in Canada once it's signed and witnessed properly, and every will comes with clear signing instructions.
"Is my clients' data safe?"Willful uses strong encryption, runs regular audits, and works with brands like Scotiabank, CIBC and Wealthsimple. Documents aren't stored on Willful's servers.
"What if an estate is complicated?"Around 80 to 90% of Canadian estates likely don't need a lawyer. For trusts, business succession or assets overseas, we'd point them to an estate lawyer.
"Is this going to replace me?"No. It gives you a reason to have the conversation, and keeps you in the middle of it.
